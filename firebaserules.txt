rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // --- CORRECTED RULE FOR LANDING PAGE ---
    // Allow anyone to create a document for email sign-ups.
    // Allow updates ONLY IF the email field is not being changed.
    // This allows adding the 'source' field without compromising the original email.
    // Allow admins to read for analytics, but no public read/delete to protect user emails.
    match /early-access-emails/{emailId} {
      allow create: if true;
      allow update: if request.resource.data.email == resource.data.email;
      allow read: if isAdmin();
      allow delete: if false;
    }

    // --- NEW RULE FOR FEEDBACK ---
    // Allow anyone to create a document for feedback submissions.
    // Only admins can read/update/delete feedback to protect user privacy.
    match /feedback/{feedbackId} {
      allow create: if true;
      allow read, update, delete: if isAdmin();
    }

    // --- NEW RULE FOR ACCOUNT DELETION REQUESTS ---
    // Allow anyone to create a document for account deletion requests.
    // Only admins can read/update/delete requests to protect user privacy.
    match /account-deletion-requests/{requestId} {
      allow create: if true;
      allow read, update, delete: if isAdmin();
    }

    // --- EXISTING APP RULES (UNCHANGED) ---

    // TEMPORARY: Admins by hardcoded email
    function isAdmin() {
      return request.auth != null &&
        (request.auth.token.email == "chrisvpopoca@gmail.com" ||
         request.auth.token.email == "adrianajaltam@gmail.com" ||
         request.auth.token.email == "cv298@njit.edu" ||
         request.auth.token.email == "israelram5112@gmail.com" ||
         request.auth.token.email == "jimmy91215@gmail.com");
    }

    function isAuthenticated() {
      return request.auth != null;
    }

    function isUser(userId) {
      return isAuthenticated() && request.auth.uid == userId;
    }

    function areFriends(otherUserId) {
      return isAuthenticated() &&
        exists(/databases/$(database)/documents/userFriends/$(request.auth.uid)) &&
        get(/databases/$(database)/documents/userFriends/$(request.auth.uid)).data.friends != null &&
        otherUserId in get(/databases/$(database)/documents/userFriends/$(request.auth.uid)).data.friends;
    }

    function isSquadMember(squadId) {
      return isAuthenticated() &&
        exists(/databases/$(database)/documents/squads/$(squadId)) &&
        get(/databases/$(database)/documents/squads/$(squadId)).data.members != null &&
        request.auth.uid in get(/databases/$(database)/documents/squads/$(squadId)).data.members;
    }

    function isSquadCreator(squadId) {
      return isAuthenticated() &&
        exists(/databases/$(database)/documents/squads/$(squadId)) &&
        get(/databases/$(database)/documents/squads/$(squadId)).data.createdBy == request.auth.uid;
    }

    function isGameCreator(gameId) {
      return isAuthenticated() &&
        exists(/databases/$(database)/documents/games/$(gameId)) &&
        get(/databases/$(database)/documents/games/$(gameId)).data.createdBy == request.auth.uid;
    }

    // Catch-all: authenticated reads
    match /{document=**} {
      allow read: if isAuthenticated();
    }

    // --- NEW --- Stories collection rules
    match /stories/{storyId} {
      // Anyone can read stories (friends can see them)
      allow read: if true;

      // Users can create their own stories
      allow create: if isAuthenticated() &&
        request.auth.uid == request.resource.data.userId &&
        request.resource.data.mediaUrl != null &&
        request.resource.data.mediaType != null &&
        request.resource.data.createdAt != null &&
        request.resource.data.expiresAt != null &&
        request.resource.data.isActive == true;

      // Users can update their own stories (for view count, viewers, etc.)
      allow update: if isAuthenticated() && (
        // Story owner can update
        request.auth.uid == resource.data.userId ||
        // Any user can update viewers and viewCount when viewing
        (request.resource.data.diff(resource.data).affectedKeys().hasOnly(['viewers', 'viewCount']) &&
         request.auth.uid in request.resource.data.viewers)
      );

      // Users can delete their own stories, admins can delete any story
      allow delete: if isAuthenticated() && (
        request.auth.uid == resource.data.userId ||
        isAdmin()
      );
    }

    // --- NEW --- User Stories collection (for quick friend story queries)
    match /userStories/{userId} {
      // Anyone can read user stories lists
      allow read: if true;

      // Users can create/update their own stories list
      allow create, update: if isAuthenticated() && request.auth.uid == userId;

      // Users can delete their own stories list
      allow delete: if isAuthenticated() && request.auth.uid == userId;
    }

    // --- NEW --- Saved Stories collection (for profile saved stories)
    match /savedStories/{savedStoryId} {
      // Users can read their own saved stories
      allow read: if isAuthenticated() && resource.data.userId == request.auth.uid;

      // Users can save stories (create)
      allow create: if isAuthenticated() &&
        request.auth.uid == request.resource.data.userId &&
        request.resource.data.storyId != null &&
        request.resource.data.originalStoryData != null &&
        request.resource.data.savedAt != null;

      // Users can delete their own saved stories
      allow delete: if isAuthenticated() && request.auth.uid == resource.data.userId;
    }

    // Games collection rules - ENHANCED for creator game management
    match /games/{gameId} {
      allow read: if true; // Anyone can read games

      // Allow creating games if authenticated and creator is self
      allow create: if isAuthenticated() &&
        request.auth.uid == request.resource.data.createdBy &&
        request.resource.data.players != null &&
        request.resource.data.players.size() > 0;

      // Allow updates for:
      // 1. Game creator (can update most fields including leaving their own game)
      // 2. Any authenticated user (for joining/leaving - adding/removing from players array)
      allow update: if isAuthenticated() && (
        // Creator can update most fields except critical ones
        (request.auth.uid == resource.data.createdBy &&
         !request.resource.data.diff(resource.data).affectedKeys().hasAny(['createdBy', 'fieldId'])) ||

        // Any user can join (add themselves to players array) or leave (remove themselves)
        (request.resource.data.diff(resource.data).affectedKeys().hasOnly(['players', 'updatedAt']) &&
         request.resource.data.createdBy == resource.data.createdBy &&
         request.resource.data.fieldId == resource.data.fieldId)
      );

      // ENHANCED: Allow deletion for creator OR admin (for game cancellation feature)
      // This allows creators to completely cancel/delete their games
      allow delete: if isAuthenticated() && (
        request.auth.uid == resource.data.createdBy ||
        isAdmin()
      );
    }

    // SIMPLIFIED Game chat messages collection - Open permissions for testing
    match /game-chats/{messageId} {
      // Anyone can read game chat messages
      allow read: if true;

      // Any authenticated user can create messages (simplified for now)
      allow create: if isAuthenticated() &&
        request.auth.uid == request.resource.data.userId &&
        request.resource.data.gameId != null &&
        request.resource.data.message != null &&
        request.resource.data.timestamp != null;

      // Users can update their own messages
      allow update: if isAuthenticated() &&
        request.auth.uid == resource.data.userId;

      // Users can delete their own messages, game creators can delete any message
      allow delete: if isAuthenticated() && (
        request.auth.uid == resource.data.userId ||
        isGameCreator(resource.data.gameId)
      );
    }

    // Archived games collection rules - FIXED for Cloud Functions
    match /archived-games/{archivedGameId} {
      // Anyone can read archived games (for stats, history, etc.)
      allow read: if true;

      // Allow creation from authenticated users OR server-side operations (Cloud Functions)
      allow create: if true;

      // No updates or deletes allowed on archived games to preserve historical data
      allow update, delete: if false;
    }

    // ---UPDATED--- Shorts collection rules
    match /shorts/{shortId} {
      allow read: if true;
      allow create: if isAuthenticated() &&
        request.auth.uid == request.resource.data.userId;
      allow update: if isAuthenticated() && (
        request.auth.uid == resource.data.userId ||
        (request.resource.data.diff(resource.data).affectedKeys()
          .hasOnly(['likes', 'likedBy', 'comments', 'pollData']))
      );
      allow delete: if isAuthenticated() &&
        (request.auth.uid == resource.data.userId || isAdmin());
    }

    // Shorts comments collection
    match /shorts-comments/{commentId} {
      allow read: if true;
      allow create: if isAuthenticated() &&
        request.auth.uid == request.resource.data.userId;
      allow update: if isAuthenticated() &&
        request.auth.uid == resource.data.userId;
      allow delete: if isAuthenticated() &&
        request.auth.uid == resource.data.userId;
    }

    // Posts/Shorts collection rules - wide open for testing
    match /posts/{postId} {
      allow read: if true;
      allow write: if isAuthenticated();

      // Also allow writing to subcollections
      match /{subcollection}/{docId} {
        allow read: if true;
        allow write: if isAuthenticated();
      }
    }

    // Feed Posts collection rules - for the new social feed
    match /feedPosts/{postId} {
      allow read: if true;
      allow create: if isAuthenticated() &&
        request.auth.uid == request.resource.data.authorId;
      allow update: if isAuthenticated() && (
        request.auth.uid == resource.data.authorId ||
        // Allow updates for likes and comments by any authenticated user
        request.resource.data.diff(resource.data).affectedKeys()
          .hasOnly(['likes', 'likesCount', 'commentsCount'])
      );
      allow delete: if isAuthenticated() && (
        request.auth.uid == resource.data.authorId ||
        isAdmin()
      );
    }

    // Feed Posts comments collection
    match /feedPosts-comments/{commentId} {
      allow read: if true;
      allow create: if isAuthenticated() &&
        request.auth.uid == request.resource.data.userId;
      allow update: if isAuthenticated() &&
        request.auth.uid == resource.data.userId;
      allow delete: if isAuthenticated() &&
        request.auth.uid == resource.data.userId;
    }

    // Shared posts collection
    match /sharedPosts/{shareId} {
      allow read: if isAuthenticated() && (
        request.auth.uid == resource.data.fromUserId ||
        request.auth.uid == resource.data.toUserId
      );
      allow create: if isAuthenticated() &&
        request.auth.uid == request.resource.data.fromUserId;
      allow update: if isAuthenticated() &&
        request.auth.uid == resource.data.toUserId &&
        request.resource.data.diff(resource.data).affectedKeys().hasOnly(['status']);
      allow delete: if isAuthenticated() && (
        request.auth.uid == resource.data.fromUserId ||
        request.auth.uid == resource.data.toUserId
      );
    }

    // Posts comments collection
    match /posts-comments/{commentId} {
      allow read: if true;
      allow create: if isAuthenticated();
      allow update: if isAuthenticated() &&
        request.auth.uid == resource.data.userId;
      allow delete: if isAuthenticated() &&
        request.auth.uid == resource.data.userId;
    }

    // ---UPDATED--- Photos collection
    match /photos/{photoId} {
      allow read: if true;
      allow create: if isAuthenticated() &&
        request.auth.uid == request.resource.data.userId;
      allow update: if isAuthenticated() &&
        request.auth.uid == resource.data.userId;
      allow delete: if isAuthenticated() &&
        (request.auth.uid == resource.data.userId || isAdmin());
    }

    // Squads collection rules
    match /squads/{squadId} {
      allow read: if isSquadMember(squadId);
      allow create: if isAuthenticated()
        && request.resource.data.createdBy == request.auth.uid
        && request.auth.uid in request.resource.data.members
        && request.resource.data.members.size() > 0
        && request.resource.data.name is string
        && request.resource.data.name.size() > 0;
      allow update: if isAuthenticated() && (
        // Squad creator can update most fields
        isSquadCreator(squadId) ||
        // Squad members can update if they're only changing the members array (for leaving)
        (isSquadMember(squadId) &&
         request.resource.data.diff(resource.data).affectedKeys().hasOnly(['members']) &&
         resource.data.createdBy in request.resource.data.members) ||
        // Squad members can update for adding/removing members while keeping creator in members
        (isSquadMember(squadId) &&
         request.resource.data.diff(resource.data).affectedKeys().hasOnly(['members', 'updatedAt']) &&
         resource.data.createdBy in request.resource.data.members)
      );
      allow delete: if isSquadCreator(squadId) ||
        // Allow deletion of direct messages by any original member
        (exists(/databases/$(database)/documents/squads/$(squadId)) &&
         get(/databases/$(database)/documents/squads/$(squadId)).data.isDirectMessage == true &&
         request.auth.uid in get(/databases/$(database)/documents/squads/$(squadId)).data.members);
    }

    // Squad chats and messages
    match /squadChats/{chatId} {
      allow read: if isAuthenticated() && (
        // For squad chats: check squad membership
        (resource.data.squadId != null && isSquadMember(resource.data.squadId)) ||
        // For direct messages: check if user is in members array
        (resource.data.type == 'dm' && request.auth.uid in resource.data.members) ||
        // Fallback: check if user is in members array (for any chat type)
        (request.auth.uid in resource.data.members)
      );
      allow create: if isAuthenticated() && (
        // For squad chats: check squad membership
        (request.resource.data.squadId != null && isSquadMember(request.resource.data.squadId)) ||
        // For direct messages: check if user is in members array
        (request.resource.data.type == 'dm' && request.auth.uid in request.resource.data.members)
      );
      allow update: if isAuthenticated() && (
        // For squad chats: check squad membership
        (resource.data.squadId != null && isSquadMember(resource.data.squadId)) ||
        // For direct messages: check if user is in members array
        (resource.data.type == 'dm' && request.auth.uid in resource.data.members) ||
        // Fallback: check if user is in members array
        (request.auth.uid in resource.data.members)
      );
      allow delete: if isAuthenticated() && (
        (resource.data.squadId != null && isSquadMember(resource.data.squadId)) ||
        (resource.data.type == 'dm' && request.auth.uid in resource.data.members) ||
        (request.auth.uid in resource.data.members)
      );

      match /messages/{messageId} {
        allow read: if isAuthenticated() && (
          exists(/databases/$(database)/documents/squadChats/$(chatId)) &&
          (
            // For squad-based chats: check if user is a squad member
            (get(/databases/$(database)/documents/squadChats/$(chatId)).data.squadId != null &&
             isSquadMember(get(/databases/$(database)/documents/squadChats/$(chatId)).data.squadId)) ||
            // For direct messages: check if user is in members array
            (get(/databases/$(database)/documents/squadChats/$(chatId)).data.type == 'dm' &&
             request.auth.uid in get(/databases/$(database)/documents/squadChats/$(chatId)).data.members) ||
            // Fallback: check if user is in members array (for any chat type)
            (request.auth.uid in get(/databases/$(database)/documents/squadChats/$(chatId)).data.members)
          )
        );
        allow create: if isAuthenticated()
          && request.resource.data.senderId == request.auth.uid
          && request.resource.data.timestamp != null
          && exists(/databases/$(database)/documents/squadChats/$(chatId))
          && (
            // For squad-based chats: check if user is a squad member
            (get(/databases/$(database)/documents/squadChats/$(chatId)).data.squadId != null &&
             isSquadMember(get(/databases/$(database)/documents/squadChats/$(chatId)).data.squadId)) ||
            // For direct messages: check if user is in members array
            (get(/databases/$(database)/documents/squadChats/$(chatId)).data.type == 'dm' &&
             request.auth.uid in get(/databases/$(database)/documents/squadChats/$(chatId)).data.members) ||
            // Fallback: check if user is in members array (for any chat type)
            (request.auth.uid in get(/databases/$(database)/documents/squadChats/$(chatId)).data.members)
          );
        allow update: if isAuthenticated()
          && exists(/databases/$(database)/documents/squadChats/$(chatId))
          && (
            // For squad-based chats: check if user is a squad member
            (get(/databases/$(database)/documents/squadChats/$(chatId)).data.squadId != null &&
             isSquadMember(get(/databases/$(database)/documents/squadChats/$(chatId)).data.squadId)) ||
            // For direct messages: check if user is in members array
            (get(/databases/$(database)/documents/squadChats/$(chatId)).data.type == 'dm' &&
             request.auth.uid in get(/databases/$(database)/documents/squadChats/$(chatId)).data.members) ||
            // Fallback: check if user is in members array (for any chat type)
            (request.auth.uid in get(/databases/$(database)/documents/squadChats/$(chatId)).data.members)
          );
        allow delete: if isAuthenticated() && (
          // User can delete their own messages
          request.auth.uid == resource.data.senderId ||
          // Chat members can delete messages (for moderation)
          (exists(/databases/$(database)/documents/squadChats/$(chatId)) &&
           (
             // For squad-based chats: check if user is a squad member
             (get(/databases/$(database)/documents/squadChats/$(chatId)).data.squadId != null &&
              isSquadMember(get(/databases/$(database)/documents/squadChats/$(chatId)).data.squadId)) ||
             // For direct messages: check if user is in members array
             (get(/databases/$(database)/documents/squadChats/$(chatId)).data.type == 'dm' &&
              request.auth.uid in get(/databases/$(database)/documents/squadChats/$(chatId)).data.members) ||
             // Fallback: check if user is in members array (for any chat type)
             (request.auth.uid in get(/databases/$(database)/documents/squadChats/$(chatId)).data.members)
           ))
        );
      }
    }

    // Users collection
    match /users/{userId} {
      allow read: if true;
      allow create: if request.auth.uid == userId;
      allow update: if request.auth.uid == userId || isAdmin();
      allow delete: if isUser(userId) || isAdmin();
    }

    // Friend requests
    match /friendRequests/{requestId} {
      allow read: if isAuthenticated();
      allow create: if isAuthenticated()
        && request.resource.data.senderId == request.auth.uid
        && request.resource.data.status == 'pending';

      // Receiver can accept/reject, sender can cancel; only status + updatedAt can change
      allow update: if isAuthenticated()
        && request.resource.data.senderId == resource.data.senderId
        && request.resource.data.receiverId == resource.data.receiverId
        && request.resource.data.diff(resource.data)
          .affectedKeys()
          .hasOnly(['status', 'updatedAt'])
        && (
          (request.auth.uid == resource.data.receiverId
            && request.resource.data.status in ['accepted','rejected'])
          || (request.auth.uid == resource.data.senderId
            && request.resource.data.status == 'canceled')
        );

      // Either party can delete
      allow delete: if isAuthenticated()
        && (request.auth.uid == resource.data.senderId
          || request.auth.uid == resource.data.receiverId);
    }

    // ---UPDATED--- User friends list - FIXED for friend acceptance permissions
    match /userFriends/{userId} {
      allow read: if isAuthenticated();

      // Allow users to create their own friends list OR other users to create it during friend operations
      allow create: if isAuthenticated() && (
        request.auth.uid == userId
        ||
        // Allow creation during friend operations if it contains only valid friend data
        (request.resource.data.keys().hasOnly(['friends', 'updatedAt']) &&
         request.resource.data.friends is list &&
         request.auth.uid in request.resource.data.friends)
      );

      // Simplified update rules
      allow update: if isAuthenticated() && (
        // Owner can freely update their own document
        request.auth.uid == userId
        ||
        // Other authenticated users can update for friend operations
        (
          // Allow updating friends array and updatedAt
          request.resource.data.diff(resource.data).affectedKeys().hasOnly(['friends', 'updatedAt'])
          &&
          // Ensure the user making the update is involved in the friendship
          (request.auth.uid in request.resource.data.friends ||
           request.auth.uid in resource.data.friends)
        )
      );

      allow delete: if request.auth.uid == userId || isAdmin();
    }

    // ---UPDATED--- User activities
    match /userActivities/{activityId} {
      allow read: if isAuthenticated()
        && (resource.data.userId == request.auth.uid
          || (resource.data.isPublic == true
            && areFriends(resource.data.userId)));
      allow create: if request.auth.uid == request.resource.data.userId;
      allow update, delete: if request.auth.uid == resource.data.userId || isAdmin();
    }

    // Admins collection
    match /admins/{adminId} {
      allow read: if isAuthenticated();
      allow create: if request.auth.uid == adminId;
      allow update, delete: if isAdmin();
    }

    // Fields and nested traffic/comments
    match /fields/{fieldId} {
      allow read: if true;
      allow create, update: if isAuthenticated();
      allow delete: if isAdmin();

      match /traffic-reports/{reportId} {
        allow read: if true;
        allow create: if isAuthenticated();
        allow update, delete: if isUser(resource.data.userId) || isAdmin();
      }

      match /field-comments/{commentId} {
        allow read: if true;
        allow create: if isAuthenticated();
        allow update, delete: if isUser(resource.data.userId) || isAdmin();
      }
    }

    // Global traffic-reports - FIXED RULE FOR COMMENTS
    match /traffic-reports/{reportId} {
      allow read: if true;
      allow create: if isAuthenticated() &&
        request.auth.uid == request.resource.data.userId &&
        request.resource.data.fieldId != null &&
        request.resource.data.timestamp != null;
      allow update, delete: if isUser(resource.data.userId) || isAdmin();
    }

    // Field traffic history
    match /field-traffic-history/{docId} {
      allow read: if true;
      allow create, update: if isAuthenticated();
      allow delete: if isAdmin();
    }

    // Field suggestions
    match /field-suggestions/{suggestionId} {
      allow create: if isAuthenticated();
      allow read: if isAuthenticated();
      allow update: if isAdmin() || (isAuthenticated() && resource.data.status == 'approved');
      allow delete: if isAdmin();
    }

    // ---UPDATED--- Field photos
    match /field-photos/{photoId} {
      allow read: if true;
      allow create: if isAuthenticated() &&
        request.auth.uid == request.resource.data.userId;
      allow update: if isAuthenticated() &&
        request.auth.uid == resource.data.userId;
      allow delete: if isAuthenticated() &&
        (request.auth.uid == resource.data.userId || isAdmin());
    }

    // ---FIXED--- Field comments collection - MAIN FIX FOR YOUR ISSUE
    match /field-comments/{commentId} {
      allow read: if true;
      allow create: if isAuthenticated() &&
        request.auth.uid == request.resource.data.userId &&
        request.resource.data.fieldId != null &&
        request.resource.data.text != null &&
        request.resource.data.timestamp != null;
      allow update, delete: if isUser(resource.data.userId) || isAdmin();
    }

    // ---NEW--- User deletion logs
    match /user-deletions/{deletionId} {
      allow read, create: if isAdmin();
      allow update, delete: if false; // Logs should be immutable
    }

    // Direct messages/chats
    match /chats/{chatId} {
      // Allow read if user is a participant in the chat
      allow read: if isAuthenticated() &&
        request.auth.uid in resource.data.participants;

      // Allow create if authenticated user is setting themselves as a participant
      allow create: if isAuthenticated() &&
        request.auth.uid in request.resource.data.participants;

      // Allow update if user is a participant and only certain fields change
      allow update: if isAuthenticated() &&
        request.auth.uid in resource.data.participants &&
        request.resource.data.diff(resource.data).affectedKeys().hasOnly(['lastMessage', 'lastMessageTimestamp', 'unreadCount']);

      // No deletion allowed
      allow delete: if false;

      // Chat messages subcollection
      match /messages/{messageId} {
        allow read: if isAuthenticated() &&
          request.auth.uid in get(/databases/$(database)/documents/chats/$(chatId)).data.participants;

        allow create: if isAuthenticated() &&
          request.auth.uid == request.resource.data.senderId &&
          request.auth.uid in get(/databases/$(database)/documents/chats/$(chatId)).data.participants;

        allow update: if isAuthenticated() &&
          request.auth.uid in get(/databases/$(database)/documents/chats/$(chatId)).data.participants &&
          request.resource.data.diff(resource.data).affectedKeys().hasOnly(['read']);

        allow delete: if isAuthenticated() &&
          request.auth.uid == resource.data.senderId;
      }
    }

    // Notifications collection rules
    match /notifications/{notificationId} {
      // Users can read their own notifications
      allow read: if isAuthenticated() && resource.data.userId == request.auth.uid;

      // Users can create notifications for any user (needed for chat messages and game notifications)
      allow create: if isAuthenticated();

      // Users can update/delete their own notifications (for marking as read or clearing)
      allow update, delete: if isAuthenticated() && resource.data.userId == request.auth.uid;
    }

    // Reports collection rules
    match /reports/{reportId} {
      // Only admins can read reports (for moderation dashboard)
      allow read: if isAdmin();

      // Any authenticated user can create reports
      allow create: if isAuthenticated() &&
        request.auth.uid == request.resource.data.reporterId &&
        request.resource.data.reportType != null &&
        request.resource.data.reason != null &&
        request.resource.data.timestamp != null &&
        request.resource.data.status == 'pending';

      // Only admins can update reports (to change status, add notes, etc.)
      allow update: if isAdmin();

      // Only admins can delete reports (for cleanup)
      allow delete: if isAdmin();
    }

    // Analytics events collection rules - for admin dashboard
    match /analytics-events/{eventId} {
      // Admins can read all analytics events for dashboard
      allow read: if isAdmin();

      // Any authenticated user can create analytics events
      allow create: if isAuthenticated();

      // Only admins can update/delete analytics events
      allow update, delete: if isAdmin();
    }

    // User sessions collection rules - for admin analytics
    match /user-sessions/{sessionId} {
      // Admins can read all user sessions for analytics
      allow read: if isAdmin();

      // Users can create their own sessions
      allow create: if isAuthenticated() &&
        request.auth.uid == request.resource.data.userId;

      // Users can update their own sessions, admins can update any
      allow update: if isAuthenticated() && (
        request.auth.uid == resource.data.userId ||
        isAdmin()
      );

      // Only admins can delete sessions
      allow delete: if isAdmin();
    }

    // App settings collection rules - for admin settings management
    match /app-settings/{settingId} {
      // Admins can read all settings
      allow read: if isAdmin();

      // Only admins can create/update/delete settings
      allow create, update, delete: if isAdmin();
    }
  }
